SELECT  ECFH_SAP_CODE,ECFH_DOC_ID,ECFH_SALES_ORG,EWDHH_CREATED_ON,EWDHH_MODIFIED_ON,
EWDHH_CREATED_BY,EWDHH_WF_STATUS,EWDHH_NEXT_PARTICIPANT,NEXT_PART FROM  
EZC_CUSTOMER_FORM_HEADER A INNER JOIN EZC_WF_DOC_HISTORY_HEADER B ON A.ECFH_DOC_ID=B.EWDHH_DOC_ID 
LEFT OUTER JOIN (SELECT EWWU_GROUP,GROUP_CONCAT(CONCAT(EU_ID,'[',EU_FIRST_NAME,']')) AS NEXT_PART  
FROM EZC_WF_WORKGROUP_USERS,EZC_USERS WHERE EWWU_USER=EU_ID AND TRIM(EWWU_USER) IN 
(SELECT EUDKV_USER_ID FROM EZC_USER_DEF_KEY_VALUES WHERE EUDKV_KEY='PURORG' AND EUDKV_VALUE IN ('1001','3001')) 
GROUP BY EWWU_GROUP) AS C ON B.EWDHH_NEXT_PARTICIPANT=C.EWWU_GROUP  WHERE A.ECFH_SALES_ORG IN ('1001','3001') 
AND B.EWDHH_WF_STATUS IN ('SUBMITTED','REJECTED','APPROVED') AND  B.EWDHH_NEXT_PARTICIPANT IN ('PUR_PERSON','VENDAPR') 
AND B.EWDHH_CREATED_ON BETWEEN STR_TO_DATE('01/04/2021 12:01','%d/%m/%Y %H:%i') AND 
STR_TO_DATE('11/11/2021 23:59','%d/%m/%Y %H:%i') 


SELECT *,(SELECT GROUP_CONCAT(CONCAT(B.EU_ID,'[',B.EU_FIRST_NAME,']')) 
FROM EZC_WF_WORKGROUP_USERS A,EZC_USERS B WHERE A.EWWU_USER=B.EU_ID  AND A.EWWU_GROUP =EWPC_GROUP 
AND TRIM(A.EWWU_USER) IN (SELECT EUDKV_USER_ID FROM EZC_USER_DEF_KEY_VALUES WHERE EUDKV_KEY='PURORG' 
AND EUDKV_VALUE='1001') AND A.EWWU_USER IN (SELECT EUDKV_USER_ID FROM EZC_USER_DEF_KEY_VALUES
) GROUP BY A.EWWU_GROUP) AS USER_STR FROM 
EZC_WF_PRICING_CONDITIONS WHERE EWPC_TEMPLATE IN (SELECT EWKM_TEMPLATE FROM EZC_WF_KEY_MAP WHERE EWKM_COMP_CODE='5000' AND EWKM_PLANT='1001' AND EWKM_CATEGORY='VNR' AND EWKM_DOC_TYPE='VNR') ORDER BY EWPC_LEVEL
